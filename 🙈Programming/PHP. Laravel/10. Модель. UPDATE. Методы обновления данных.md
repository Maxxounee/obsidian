#### PostController

```php
<?php

namespace App\Http\Controllers;

use App\Models\Post;
use Illuminate\Http\Request;

class PostController extends Controller
{
    public function update() {
        $post = Post::find(5);

        $post->update([
            'title'=>'updated',
        ]);
    }
}
```
