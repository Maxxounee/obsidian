```php
class Post extends Model {
    use SoftDeletes;
    protected $table = 'posts';
    protected $guarded = ['id'];
    
    protected $fillable = [];

	# один ко многим
    public function category() {
        return $this->belongsTo(
            Category::class,
            'category_id',
            'id'
        );
    }

    /* Многие ко многим */
    public function tags() {
        return $this->belongsToMany(
            Tag::class, # С какой таблицей связывается Post (related)
            'post_tags', # Через что (table. многие ко многим)
            'post_id', # Id данной таблицы (foreignPivotKey)
            'tag_id' # чужой id (relatedPivotKey)
        );
    }
```
