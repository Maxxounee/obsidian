#### PostController

```php
    public static function store() {
        $data = request()->validate([
            'title' => 'string',
            'content' => 'string',
            'image' => 'string',
            'category_id' => 'integer',
            'tags' => 'array',
        ]);

        $tags = $data['tags'];
        unset($data['tags']);

        $post = Post::create($data);

        if (false) {
            # Вариант добавить tags колхозный
            foreach ($tags as $tag) {
                /* firstOrCreate Создаст 
                только если такого сочетания нет в БД */
                PostTag::firstOrCreate([
                    'tag_id' => $tag,
                    'post_id' => $post->id,
                ]);
            }
        }

        # Вариант добавить tag нормальный
        $post->tags()->attach($tags);

        return redirect()->route('posts.index');
    }
```
