#### PostController.php

```php
public function edit($id) {
        $post = Post::findOrFail($id);
        $categories = Category::all();
        $tags = Tag::all();

        return view(
            'posts.edit',
            compact([
                'post',
                'id',
                'categories',
                'tags'
            ])
        );
    }

public function update(Post $post) {
        $data = request()->validate([
            'title' => 'string',
            'content' => 'string',
            'image' => 'string',
            'category_id' => '',
            'tags' => 'array',
        ]);

        $tags = $data['tags'];
        unset($data['tags']);

        $post->update($data);
        $post->tags()->sync($tags);

        return redirect()->route('posts.show', $post->id);```
