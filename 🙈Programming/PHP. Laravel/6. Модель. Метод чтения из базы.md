#### app/http/controllers/PostController.php

```php
<?php

namespace App\Http\Controllers;

use App\Models\Post;
use Illuminate\Http\Request;

class PostController extends Controller
{
    public static function index() {
        // ::find так же имплементирует (как бы) создание класса new Post.
        // Поэтому в переменную $post сохраняются свойства
        // класса Post (extends Model)
        $post = Post::find(1);

        # Получить значение "title" по ключу attributes
        # (laravel использует магический метод __get($key))
        dump($post->title);
        dump($post->getAttributes());

        /* Покажет все свойства (в т.ч. и приватные) Post (extends Model) */
        dump($post);
        /* Значение ключа $post->table (приватное свойство. Доступ
        только через публичные методы) */
        dump($post->getTable());
        /* Значение ключа $post->connection */
        dump($post->getConnectionName());

        return 'this is my page';
    }
}
```
