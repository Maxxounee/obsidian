```bash
# Создаем проект
composer create-project laravel/laravel example_app

# Установка vite
composer require laravel/ui

php artisan ui bootstrap
php artisan ui vue # или
php artisan ui react # или

npm i

php artisan make:controller PostController

php artisan make:model Post -m # модель + миграция

php artisan serve

npm run dev
```
