### Пример 1

```php
class Queue {
    private $ben = [];
    private $no_ben = [];

	// В функции. !Через тернарник передача по ссылке не сработает!
    private function &get_right($term) {
        if ((bool) $term) {
            return  $this->ben;
        } else {
            return $this->no_ben;
        }
    }
    
    public function add($name, $is_ben=0) {
	    // и в присваивании
        $arr = &$this->get_right($is_ben);
        $arr[] = $name;
    }
    
    public function del() {
	    // и в присваивании
        $arr = &$this->get_right(count($this->ben));
        
        echo array_shift($arr) . PHP_EOL;
    }
    
    public function count() {
        echo count($this->ben) + count($this->no_ben) . PHP_EOL;
    }
}

$people = new Queue();

$n = readline();

for ($i = 0; $i < $n; $i++) {
    [$func, $name, $is_ben] = explode(' ', readline());
    
    $people->$func($name, $is_ben);
}

$people->count();
```

### Пример 2

Тут тоже немного ссылок

```php
<?php
class TranslateHelper {
    public static function translate($word="") {
        return [
            'ros'=>'roz',
            'pover'=>'power',
        ][$word] ?? $word;
    }
}

class Schema {
    private $el = [];
    private $chLinkCount = 0;
    
    public function __construct($power="220v") {
        $this->el["power"] = [$power];
    }
    
    public function add($name, $item, $link) {
        $this->el[$name] = [
            "item"=>$item, 
            "link"=>$link,
        ];
    }
    
    public function del($name) {
        $replace_link = $this->el[$name]["link"];

		// в функции
        $this->el = array_map(function(&$val) use ($name, $replace_link) {
            
            if ($val["link"] == $name) {
                $val["link"] = $replace_link;
                // использование this внутри функции работает нормально.
                $this->chLinkCount++;
            }
            
            return $val;
        }, $this->el);
        
        unset($this->el[$name]);
    }
    
    public function getChLinksCount() {
        return $this->chLinkCount;
    }
}

$n = readline();
$electro = new Schema();

for ($i = 0; $i < $n; $i++) {
    [$func, $name, $item, $link] = explode(' ', readline());
    $item = TranslateHelper::translate($item);
    $link = TranslateHelper::translate($link);
    
    $electro->$func($name, $item, $link);
}

echo $electro->getChLinksCount();
?>
```

Но лучше будет так:

```php
public function del($name) {
    $replace_link = $this->el[$name]["link"];
    $counter = &$this->chLinkCount;  // ссылка на свойство
    
    $this->el = array_map(function(&$val) use ($name, $replace_link, &$counter) {
        //                                                            ^^^^^^^^
        if ($val["link"] == $name) {
            $val["link"] = $replace_link;
            $counter++;  // ✅ работаем через ссылку
        }
        return $val;
    }, $this->el);
    
    unset($this->el[$name]);
}
```
