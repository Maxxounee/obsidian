

```php
$config = [ // одномерный
  'user' => 'petya_12',
  'e-mail' => 'petya_2012@gmail.com',
  'url' => 'https://stepik.org/edit-lesson/1276605/step/1',
];

$config = [ // многомерный
  'user_info' => [
    'user' => 'petya_12',
    'e-mail' => 'petya_2012@gmail.com',
    'year' => 2012,
  ],
  'site' => [
    'url' => 'https://stepik.org/edit-lesson/1276605/step/1',
    'status' => 1,
    'statuses' => [
      0 => 'undefined',
      1 => 'work',
      2 => 'not work',
    ];
  ],
];
```

Перебор

```php
// foreach (массив as элемент массива) {тело цикла}
foreach ($config as $param) {
  echo $param . PHP_EOL;  
}

// foreach (массив as ключ => значение) {тело цикла}
foreach ($config as $key => $value) {  
  echo $key . ': ' . $value . PHP_EOL;  
}
```
