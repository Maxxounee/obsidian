Для каждого автора из таблицы author вывести количество книг, написанных им в каждом жанре.

Вывод: ФИО автора, жанр, количество. Отсортировать по фамилии, затем - по убыванию количества написанных книг.

```sql
/* Как смог.......... */
CREATE TABLE auth_genre_count
SELECT
    name_author,
    name_genre,
    IF(COUNT(*) IS NULL, 0, COUNT(*)) AS Количество
FROM 
    author
    LEFT JOIN book USING(author_id)
    LEFT JOIN genre USING(genre_id)
GROUP BY
    name_author,
    name_genre
;

CREATE TABLE all_genre
SELECT
    name_author,
    genre.name_genre
FROM
    auth_genre_count
    CROSS JOIN genre
;

SELECT 
    a.name_author,
    a.name_genre,
    IF(Количество IS NULL, 0, Количество) AS Количество
FROM
    all_genre AS a 
    LEFT JOIN auth_genre_count AS b 
    ON (a.name_author, a.name_genre) = (b.name_author, b.name_genre)
ORDER BY
    a.name_author,
    Количество DESC

    COUNT(DISTINCT title) AS Количество

ORDER BY name_author, Количество DESC
```
