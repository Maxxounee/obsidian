В таблицу `buy_book` добавить заказ с номером 5. Этот заказ должен содержать книгу Пастернака «Лирика» в количестве двух экземпляров и книгу Булгакова «Белая гвардия» в одном экземпляре.

```sql
INSERT INTO buy_book (buy_id, book_id, amount)
SELECT
    5,
    book_id,
    2
FROM
    book
    JOIN author USING(author_id)
WHERE
    title = "Лирика"
    AND name_author LIKE "%Пастернак%"
;

INSERT INTO buy_book (buy_id, book_id, amount)
SELECT
    5,
    book_id,
    1
FROM
    book
    JOIN author USING(author_id)
WHERE
    title = "Белая гвардия"
    AND name_author LIKE "%Булгаков%"
;

SELECT * FROM buy_book
    

/* UNION */
insert into buy_book (buy_id, book_id, amount)
	select 5, book_id, 2
	from book 
	where title = 'Лирика' 
	    and author_id = (select author_id from author where name_author = 'Пастернак Б.Л.')
union 
select 5, book_id, 1
from book 
where title = 'Белая гвардия' 
    and author_id = (select author_id from author where name_author = 'Булгаков М.А.');
  
select * from buy_book;
```
