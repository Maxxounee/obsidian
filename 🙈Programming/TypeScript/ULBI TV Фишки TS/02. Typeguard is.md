```ts
type CarBrand = 'bmw' | 'lada' | 'toyota';  
  
interface CarBase {  
    year: number;  
    brand: CarBrand;  
}  
  
interface BmwBase extends CarBase {  
    climateControl: boolean;  
    brand: 'bmw';  
    model: 'x5' | 'x7';  
}  
  
interface Bmwx5 extends BmwBase {  
    value: string;  
    model: 'x5';  
}  
  
interface Bmwx7 extends BmwBase {  
    value: string;  
    model: 'x7';  
}  
  
  
type Bmw = Bmwx5 | Bmwx7;  
type Car = Bmw;  
  
function isBmw(car: Car): car is Bmw {  
    return car.brand === 'bmw';  
}  
function assertBmw(car: any): asserts car is Bmw {  
    if (car === null || typeof car !== "object") {  
        throw new Error('Car must be an object');  
    }  
    if (car.brand !== 'bmw') {  
        throw new Error('Car model must be is BMW');  
    }  
}  
  
function fn(car: Car) {  
    // После проверки car is Bmw TS подхватит все типы.  
    if (isBmw(car)) {  
        console.log(car.year);  
        console.log(car.brand);  
        console.log(car.model);  
        console.log(car.value);  
    }  
}  
  
const someCar = {year: 123, brand: 'bmw', model: 'x5', value: 'test'};  
  
fn(someCar as Car);  

// Можно использовать asserts value is
assertBmw(someCar);  
  
fn(someCar);
```
