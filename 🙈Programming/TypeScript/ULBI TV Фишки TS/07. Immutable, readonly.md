interface User {  
    username: string;  
}  
  
const obj: Readonly<User> = {  
    username: '123',  
}  

// Тут ошибка, потому что указали readonly у obj
obj.username = 'wwww';  
  
function userFn(user: Readonly<User>) {  
	// Тут для ошибки Readonly у obj не сработает. Нужно readonly у типа User
    user.username = '321';  
}  
  
userFn(obj);
