Type guard — это любой механизм, который позволяет сузить типы. Тут поговорим о пользовательских type guards.

Создаем функцию. Должна быть реализована какая-нибудь проверка. ФУнкция должна возвращать boolean

```ts
interface Car {  
    maxSpeed: number;  
    width: number;  
}  
  
interface Person {  
    age: number;  
    name: string;  
}  
  
function isCar(value: Object): value is Car {  
    return 'maxSpeed' in value && 'width' in value;  
}  
  
function isPerson(value: any): value is Person {  
    return 'age' in value && 'name' in value;  
}  
  
function fn(data: Car | Person) {  
    if (isCar(data)) {  
        // Благодаря конструкции fn(): value is <type> TS здесь подхватит тип 
        // после выполнения функции isCar.        
        // Если сделать fn(): boolean, то TS не подхватит тип у data.        
        return data.width;  
    } else if (isPerson(data)) {  
        return data.age;  
    }  
}  
  
const car: Car = {  
    maxSpeed: 123,  
    width: 312,  
}  
  
const noCar = {};  
  
const noCar2 = {  
    maxSpeed: 123,  
    width: 312,  
}  
  
console.log(isCar(car));  // true
console.log(isCar(noCar));  // false
console.log(isCar(noCar2));  // true
console.log(isCar('asd')); // error
```
