Перегрузка функции — это когда мы можем для одной реализации объявить несколько сигнатур (функций с разными аргументами)

```ts
function makeDate(timestamp: number): Date;  
function makeDate(m: number, d: number, y: number): Date;  
function makeDate(mOrTimestamp: number, d?: number, y?: number): Date {  
    if (mOrTimestamp > 30 && d !== undefined && y !== undefined) {  
        return new Date(y, mOrTimestamp, d);  
    } else {  
        return new Date(mOrTimestamp);  
    }  
}  
  
const d1 = makeDate(12345678);  
const d2 = makeDate(5,5,5);  
const d3 = makeDate(112341, 3, 6, 3);  
  
  
function parseInput(input: string): string[];  
function parseInput(input: number): number;  
function parseInput(input: string | number): string[] | number {  
    if (typeof input === "string") {  
        return  input.split(" ");  
    }  
      
    return input * 2;  
}  
  
const p1 = parseInput(123);  
const p2 = parseInput("123 123");
```
