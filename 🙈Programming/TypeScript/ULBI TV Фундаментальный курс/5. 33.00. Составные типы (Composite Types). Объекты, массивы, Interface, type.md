
Составные типы — основа основ, самая важная часть TS.

Существует два способа описания составных типов: interface и type. В _базовом_ понимании можно считать, что они являются одним и тем же. Позже рассмотрим их отличия

```js
interface Address {
	city?: string;
	street?: string;
	coords: number[];
}

type User = {
	firstName: string;
	age?: number;
	address: Address;
}

const user: User = {
	firstName: 'Max',
	address: {
		coords: [5, 5]
	}
}

const users: User[] = [
	{
		firstName: 'Max',
		address: {
			coords: [5, 5]
		}
	}
]

function fn(userData: User) {...}

type ApiResponse<T> = {
	status: 'success' | 'error';
	data?: T;
}

type onClick = (event: HTMLevent) => void;
```
