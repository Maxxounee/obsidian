
Можно указать дефолтный тип:

```ts
interface ApiResponse<Data = string> {
	status: 'err' | 'success';
	id: string;
	data: Data;
}

const response: ApiResponse<User> = {...}; // ok
const response: ApiResponse = {...}; // тоже ok. При наличии дефолтного значения, дженерик можно не передавать									 
```

Условные типы:

```ts
// работает как тернарник
interface ApiResponse<Data extends User ? {} : {}> {
	status: 'err' | 'success';
	id: string;
	data: Data;
}
```

```ts
type isArray<T> = T extends any[] ? true : false;

const first:isArray<string> = true; // ошибка. Type 'true' is not assignable to type 'false'.
const first:isArray<string> = false; // ок

const second:isArray<string[]> = true; // ok
const second:isArray<string[]> = false; // fail. Type 'false' is not assignable to type 'true'.
```

```ts
type RandomName<T> = T extends string ? {value: string} : {value: T}

const third: RandomName<string> = { value: 'sss' }; // ok 
const third2: RandomName<Date> = { value: new Date() }; // ok
```

```ts
type User = {
	username: string;
}
type Foo = {
	value: number;
}

type RandomName<T> = T extends User ? { value: number } : { value: string }

const fourth: RandomName<User> = { value: 123 };
const fourth2: RandomName<Foo> = { value: '123' };
```
