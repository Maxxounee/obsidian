### Способ 1. typeof type guards. 

#### typeof 

Классические проверки через typeof. Работает с примитивами

```ts
function fn(arg: number | string | null) {
	if (typeof arg === 'number') {
		return;
	} else if (typeof arg === 'string') {
		return;
	} 

	return arg;
}
```

### Способ 2. Truthiness narrowing

#### Через сравнение аргументов

```ts
function fn(arg: number | string | null, arg2: number) {
	if (arg === null) {
		return;
	}

	if (arg === arg2) {
		// ts определит, что arg типа number
		return;
	}
	
	return arg;
}
```

#### Через поле объекта типа \<type>

```ts
interface User {
	username: string;
	age: number;
}

interface Person {
	lastname: string;
	firstname: string;
	age: number;
}

function fn(arg: User | Person) {
	if ('username' in arg) {
		// ts сузит тип внутри этого условия. Будет считать arg типом User
		return arg.username;
	} else if ('lastname' in arg) {
		return arg.lastname;
	}
}
```

#### Для class. Instanceof

```ts
class Bmw {  
    public model: string;  
    private year: number;  
    
        constructor(model: string, year: number) {  
	        this.model = model;  
	        this.year = year;  
	    }  
}  
  
class Audi {  
    public wheels: number = 4;  
  
    constructor(wheels: number = 4) {  
        this.wheels = wheels;  
    }  
}  
  
function fn(arg: Bmw | Audi) {  
    if (arg instanceof Bmw) {  
        return arg.model;  
    } else if (arg instanceof Audi) {  
        return arg.wheels;  
    }  
}
```

#### interface

В интерфейс необходимо добавить поле, которое однозначно будет идентифицировать тип (`type: 'bmw'`)

```ts
interface BaseCar {  
    maxSpeed: number;  
    weight: number;  
}  
  
interface Bmw extends BaseCar {  
    type: 'bmw';  
    bwmField: string;  
}  
  
interface Audi extends BaseCar {  
    type: 'audi';  
    audiField: string;  
}  
  
type Car = Audi | Bmw;  
  
function fn(arg: Bmw | Audi) {  
	// ts по одному полю type может сузить тип аргумента
    switch(arg.type) {  
        case "bmw":  
            break;  
        case "audi":  
            break;  
        default:
	        // тут будет never.
    }  
}
```
